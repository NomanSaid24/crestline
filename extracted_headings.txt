(function () {
      var classGuard = new MutationObserver(function () {
        if (!document.documentElement.classList.contains('wm-loading')) {
          document.documentElement.classList.add('wm-loading');
        }
      });
      classGuard.observe(document.documentElement, { attributes: true, attributeFilter: ['class'] });

      var BG = '#fef073';
      var EASE = 'cubic-bezier(0.76, 0, 0.24, 1)';

      var loader = document.createElement('div');
      loader.id = 'wmLoader';
      loader.style.cssText = [
        'position:fixed', 'inset:0', 'z-index:9999999',
        'background:' + BG, 'width:100vw', 'height:100vh',
        'overflow:hidden', 'pointer-events:none'
      ].join(';');

      var bar = document.createElement('div');
      bar.style.cssText = 'background:#000;width:0.35rem;height:0;position:absolute;bottom:0;right:0;z-index:5;';
      loader.appendChild(bar);

      var maskTop = document.createElement('div');
      maskTop.style.cssText = [
        'position:absolute', 'top:0', 'left:0', 'right:0',
        'background:' + BG, 'z-index:3',
        'height:100vh'
      ].join(';');
      loader.appendChild(maskTop);

      var maskBtm = document.createElement('div');
      maskBtm.style.cssText = [
        'position:absolute', 'bottom:0', 'left:0', 'right:0',
        'background:' + BG, 'z-index:3',
        'height:0'
      ].join(';');
      loader.appendChild(maskBtm);

      var wrapper = document.createElement('div');
      wrapper.style.cssText = [
        'position:absolute', 'top:0', 'left:2em', 'right:2em',
        'z-index:2', 'will-change:transform',
        'transform:translateY(100vh)'
      ].join(';');

      var lineEls = [];
      ['CONTENT +', 'SOCIAL MEDIA +', 'PAID MEDIA'].forEach(function (txt) {
        var mask = document.createElement('div');
        mask.style.cssText = 'display:block;overflow:hidden;';

        var p = document.createElement('div');
        p.textContent = txt;
        p.style.cssText = [
          'display:block',
          'font-family:"Neue Haas Display","Neue Haas Grotesk Display Web",Arial,Helvetica,sans-serif',
          'font-size:5.2vw',
          'font-weight:700',
          'line-height:0.92',
          'letter-spacing:-0.03em',
          'color:#000',
          'text-transform:uppercase',
          'white-space:nowrap',
          'margin:0',
          'transform:translateY(110%)',
          'will-change:transform'
        ].join(';');

        mask.appendChild(p);
        wrapper.appendChild(mask);
        lineEls.push(p);
      });

      loader.appendChild(wrapper);

      var done = false;
      function ensureLoader() {
        if (done) return;
        if (!document.getElementById('wmLoader')) document.documentElement.appendChild(loader);
      }

      document.documentElement.appendChild(loader);
      var guardInterval = setInterval(ensureLoader, 50);

      var TRAVEL = 480;
      var RL_DUR = 480;
      var EL_DUR = 260;
      var STAG = 60;
      var EL_TOTAL = EL_DUR + 2 * STAG;
      var RL_TOTAL = RL_DUR + 2 * STAG;
      var HOLD = 200;
      var EASE_OUT = 'cubic-bezier(0.22, 1, 0.36, 1)';
      var EASE_IN = 'cubic-bezier(0.55, 0, 1, 0.45)';

      function revealLines() {
        lineEls.forEach(function (el, i) {
          el.animate(
            [{ transform: 'translateY(110%)' }, { transform: 'translateY(0%)' }],
            { duration: RL_DUR, delay: i * STAG, easing: EASE_OUT, fill: 'forwards' }
          );
        });
      }

      function exitLinesUp() {
        lineEls.forEach(function (el, i) {
          el.animate(
            [{ transform: 'translateY(0%)' }, { transform: 'translateY(-110%)' }],
            { duration: EL_DUR, delay: i * STAG, easing: EASE_IN, fill: 'forwards' }
          );
        });
      }

      setTimeout(function () {
        var VH = window.innerHeight;
        var ZONE = VH / 3;
        var fsize = window.innerWidth * 0.052;
        var lh = fsize * 0.92;
        var H = lh * 3;

        var yBelow = VH + 20;
        var yBottom = Math.max(2 * ZONE + (ZONE - H) / 2, 2 * ZONE);
        var yCenter = Math.max(ZONE + (ZONE - H) / 2, ZONE);
        var yTop = Math.max((ZONE - H) / 2, 0);
        var yAbove = -(H + 20);

        function tx(y) {
          return 'translateY(' + y.toFixed(1) + 'px)';
        }

        wrapper.animate(
          [{ transform: tx(yBelow) }, { transform: tx(yBottom) }],
          { duration: TRAVEL, easing: EASE, fill: 'forwards' }
        );
        maskTop.animate(
          [{ height: VH + 'px' }, { height: (2 * ZONE) + 'px' }],
          { duration: TRAVEL, easing: EASE, fill: 'forwards' }
        );
        revealLines();

        var t1 = TRAVEL + RL_TOTAL + HOLD;

        setTimeout(function () {
          exitLinesUp();

          setTimeout(function () {
            wrapper.animate(
              [{ transform: tx(yBottom) }, { transform: tx(yCenter) }],
              { duration: TRAVEL, easing: EASE, fill: 'forwards' }
            );
            maskTop.animate(
              [{ height: (2 * ZONE) + 'px' }, { height: ZONE + 'px' }],
              { duration: TRAVEL, easing: EASE, fill: 'forwards' }
            );
            maskBtm.animate(
              [{ height: '0px' }, { height: ZONE + 'px' }],
              { duration: TRAVEL, easing: EASE, fill: 'forwards' }
            );
            setTimeout(revealLines, TRAVEL);
          }, EL_TOTAL);

          var t2 = EL_TOTAL + TRAVEL + RL_TOTAL + HOLD;

          setTimeout(function () {
            exitLinesUp();

            setTimeout(function () {
              wrapper.animate(
                [{ transform: tx(yCenter) }, { transform: tx(yTop) }],
                { duration: TRAVEL, easing: EASE, fill: 'forwards' }
              );
              maskTop.animate(
                [{ height: ZONE + 'px' }, { height: '0px' }],
                { duration: TRAVEL, easing: EASE, fill: 'forwards' }
              );
              maskBtm.animate(
                [{ height: ZONE + 'px' }, { height: (2 * ZONE) + 'px' }],
                { duration: TRAVEL, easing: EASE, fill: 'forwards' }
              );
              setTimeout(revealLines, TRAVEL);
            }, EL_TOTAL);

            var t3 = EL_TOTAL + TRAVEL + RL_TOTAL + HOLD;

            setTimeout(function () {
              exitLinesUp();

              setTimeout(function () {
                wrapper.animate(
                  [{ transform: tx(yTop) }, { transform: tx(yAbove) }],
                  { duration: TRAVEL, easing: EASE, fill: 'forwards' }
                );
                loader.animate(
                  [{ transform: 'translateY(0)' }, { transform: 'translateY(-100%)' }],
                  { duration: TRAVEL + 100, easing: EASE, fill: 'forwards' }
                );
                setTimeout(function () {
                  done = true;
                  classGuard.disconnect();
                  clearInterval(guardInterval);
                  loader.style.display = 'none';
                  document.documentElement.style.background = '';
                  document.documentElement.classList.remove('wm-loading');
                }, TRAVEL + 200);
              }, EL_TOTAL);
            }, t3);
          }, t2);
        }, t1);

        var totalMs = 900 + t1 +
          2 * (EL_TOTAL + TRAVEL + RL_TOTAL + HOLD) +
          EL_TOTAL + TRAVEL + 300;

        bar.animate(
          [{ height: '0' }, { height: '100vh' }],
          { duration: totalMs, easing: 'linear', fill: 'forwards' }
        );
      }, 900);
    })();
